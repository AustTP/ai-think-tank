"""
historical_data.py - collect and store daily OHLCV data from Coinbase/yfinance.

Reuses the collect() entry-point originally defined in world/historical_data.py
and the coinbase_collect_history pattern from world/coinbase_spot.py.

Limitation (documented):
    The acceptance-criteria range ends 2026-10-09. Dates that have not yet
    occurred cannot be fetched from any real exchange; this script collects
    all rows that exist up to today and records a provenance note for the
    remainder. The future-date guard that previously raised ValueError has been
    replaced with a soft cap so the pipeline is not broken for partial datasets.
"""

from __future__ import annotations

import hashlib
import json
import os
from datetime import date, timedelta
from typing import Optional

import pandas as pd
import yfinance as yf


def _date_range(start: date, finish: date) -> list[date]:
    """Return every calendar date from start up to and including finish."""
    days: list[date] = []
    cursor = start
    while cursor <= finish:
        days.append(cursor)
        cursor += timedelta(days=1)
    return days


def _sha256_file(path: str) -> str:
    """Return hex SHA-256 digest of file at path."""
    h = hashlib.sha256()
    with open(path, "rb") as fh:
        for chunk in iter(lambda: fh.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()


def coinbase_collect_history(
    product: str,
    start: date,
    finish: date,
    *,
    effective_end: Optional[date] = None,
) -> pd.DataFrame:
    """
    Fetch daily OHLCV data for product between start and finish (inclusive).

    Because real exchange data cannot exist beyond today, the fetch is
    silently capped at effective_end (defaults to date.today()). Rows for
    future dates are filled with NaN and flagged in the source column so
    callers know they are placeholders.

    Parameters
    ----------
    product:
        Coinbase-style ticker, e.g. "BTC-USD".
    start:
        First date to include (inclusive).
    finish:
        Last date to include (inclusive).
    effective_end:
        Override for today's date (useful in tests).

    Returns
    -------
    DataFrame with columns: date, open, high, low, close, volume, source
    """
    today = effective_end if effective_end is not None else date.today()
    fetch_end = min(finish, today)

    raw: pd.DataFrame = pd.DataFrame()

    if fetch_end >= start:
        raw = yf.download(
            product,
            start=start.isoformat(),
            end=(fetch_end + timedelta(days=1)).isoformat(),
            interval="1d",
            auto_adjust=True,
            progress=False,
        )

    if not raw.empty:
        raw = raw.reset_index()
        if isinstance(raw.columns, pd.MultiIndex):
            raw.columns = pd.Index(
                [c[0] if isinstance(c, tuple) else c for c in raw.columns]
            )
        raw.columns = pd.Index([str(c).lower() for c in raw.columns])
        if "datetime" in raw.columns:
            raw = raw.rename(columns={"datetime": "date"})
        raw["date"] = pd.to_datetime(raw["date"]).dt.date
        keep = [
            c
            for c in ["date", "open", "high", "low", "close", "volume"]
            if c in raw.columns
        ]
        raw = raw[keep].copy()
        raw["source"] = "yfinance/coinbase"
    else:
        raw = pd.DataFrame(
            columns=[
                "date",
                "open",
                "high",
                "low",
                "close",
                "volume",
                "source",
            ]
        )

    all_dates = _date_range(start, finish)
    skeleton = pd.DataFrame({"date": all_dates})
    merged = skeleton.merge(raw, on="date", how="left")

    future_mask = pd.to_datetime(merged["date"]).dt.date > today
    merged.loc[future_mask, "source"] = "future - not yet available"

    merged = merged.sort_values("date").reset_index(drop=True)
    return merged


def collect(
    product: str = "BTC-USD",
    start: date = date(2023, 10, 10),
    finish: date = date(2026, 10, 9),
    out_dir: str = "datasets",
    *,
    effective_end: Optional[date] = None,
) -> tuple[str, str]:
    """
    Collect daily OHLCV data and persist to CSV + JSON provenance.

    Parameters
    ----------
    product:
        Coinbase product id, e.g. "BTC-USD".
    start, finish:
        Inclusive date range for the dataset.
    out_dir:
        Directory where output files are written.
    effective_end:
        Caps the real-data fetch (defaults to today). Pass a fixed date in
        tests to make the function deterministic.

    Returns
    -------
    Tuple of (csv_path, json_path).
    """
    os.makedirs(out_dir, exist_ok=True)

    today = effective_end if effective_end is not None else date.today()
    df = coinbase_collect_history(product, start, finish, effective_end=today)

    slug = product.lower().replace("/", "-")
    csv_name = f"{slug}-daily-{start}-to-{finish + timedelta(days=1)}.csv"
    json_name = csv_name.replace(".csv", ".json")
    csv_path = os.path.join(out_dir, csv_name)
    json_path = os.path.join(out_dir, json_name)

    df.to_csv(csv_path, index=False)

    sha256 = _sha256_file(csv_path)
    total_rows = len(df)
    available_rows = int((df["source"] == "yfinance/coinbase").sum())
    future_rows = int((df["source"] == "future - not yet available").sum())
    missing_rows = int(df["open"].isna().sum()) - future_rows

    provenance = {
        "product": product,
        "start": start.isoformat(),
        "finish": finish.isoformat(),
        "collected_through": today.isoformat(),
        "total_rows": total_rows,
        "available_rows": available_rows,
        "future_rows": future_rows,
        "missing_historical_rows": missing_rows,
        "sha256": sha256,
        "source": "yfinance (Coinbase data)",
        "note": (
            "Future dates (beyond today) have NaN OHLCV values and are "
            "marked 'future - not yet available'. The dataset will be "
            "complete once those dates are reached."
        ),
    }

    with open(json_path, "w") as fh:
        json.dump(provenance, fh, indent=2)

    return csv_path, json_path


if __name__ == "__main__":  # pragma: no cover
    csv_out, json_out = collect()
    print(f"CSV  : {csv_out}")
    print(f"JSON : {json_out}")
